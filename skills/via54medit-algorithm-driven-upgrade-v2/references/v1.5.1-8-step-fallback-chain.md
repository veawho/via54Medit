# v1.5.1 — 8-Step Fallback Chain (Chrome PDF Download)

## 完整 8 步 fallback (用户硬要求顺序)

**触发**: 拿到 DOI, 需要下载真 publisher PDF.

```
Step 1: doi.org 入口 + 30 轮 Unpaywall iframe 等待 (multi-round, 每轮 1s)
   ↓ green/bronze/gold color
Step 2: Unpaywall OA URL (iframe.name 解析, 不是 iframe.contentWindow.name 因 SOP)
   ↓ green/bronze/gold
Step 3: meta_citation_pdf_url (DOM) + ASCO /doi/abs/ → /doi/pdfdirect/ 转换
   ↓
Step 4: ASCO pdfdirect shortcut (doi.startswith('10.1200/'))
   ↓
Step 5: Frontiers shortcut (FRONTIERS_SLUGS 映射 + /articles/{slug}/pdf)
   ↓
Step 6a: Unpaywall API (api.oadoi.org/v2/<doi>) + best_oa_location.url_for_pdf
Step 6b: Europe PMC (?pdf=render via Europe PMC API, safe_epmc_pdf)
   ↓
Step 7: Sci-Hub mirrors 轮询 (sci-hub.ee / .ren / .st / .sg / .box / .wf)
         监听 JS_GET_SCIHUB_PDF iframe PDF URL → fetch_via_cdp
   ↓
Step 8: Google filetype:pdf <query> 多轮 (用户硬要求, 最后手段)
```

**任一步成功 → 立即 return, 不走后续**.

## Step 1: doi.org + Unpaywall iframe (multi-round)

```python
tab.send('Page.navigate', {'url': f'https://doi.org/{doi}'})
time.sleep(8)  # 等页面 + Unpaywall content script 注入

# 30 轮 1s 轮询, 而不是 1 次 30s
findings = {}
upw_color = None
for attempt in range(30):
    time.sleep(1)
    scan_str = tab.eval(JS_SCAN)  # v13 的 JS_SCAN
    if scan_str:
        findings = json.loads(scan_str) if isinstance(scan_str, str) else {}
        upw_color = findings.get('unpaywall_color')
        if upw_color is not None:
            break  # 拿到 color 就退出
```

**JS_SCAN** (v13 已有, 关键是 `iframe.name` 属性):
```javascript
let upw = document.querySelector('iframe#unpaywall, iframe[src*="unpaywall"]');
if (upw) {
    let nm = upw.name || '';  // ← 关键: name 属性, 不是 contentWindow.name
    if (nm.includes('#')) {
        let parts = nm.split('#');
        findings.unpaywall_color = parts[0];  // green/bronze/gold/black
        if (parts[0] !== 'black') {
            findings.unpaywall_pdf_url = decodeURIComponent(parts.slice(1).join('#'));
        }
    }
}
```

## Step 6: Unpaywall API + Europe PMC (None-safe)

```python
# Step 6a: Unpaywall API (api.oadoi.org)
def safe_unpaywall_api(doi):
    """v13 unpaywall_api 的 None-safe 版本"""
    try:
        url = f'https://api.oadoi.org/v2/{doi}?email=hermes@unpaywall-test.org'
        req = urllib.request.Request(url, headers={'User-Agent': 'Mozilla/5.0'})
        with urllib.request.urlopen(req, timeout=8) as r:
            data = json.loads(r.read())
    except Exception:
        return []
    candidates = []
    best = data.get('best_oa_location') or {}  # ← None-safe
    if best.get('url_for_pdf'):
        candidates.append(('best_oa_pdf', best['url_for_pdf']))
    elif best.get('url'):
        candidates.append(('best_oa_landing', best['url']))
    for loc in (data.get('oa_locations') or []):  # ← None-safe
        if loc.get('url_for_pdf'):
            candidates.append((f"loc_{loc.get('host_type','?')}_pdf", loc['url_for_pdf']))
    return candidates

# Step 6b: Europe PMC
def safe_epmc_pdf(doi):
    """v13 epmc_pdf 的 None-safe 版本"""
    try:
        url = f'https://www.ebi.ac.uk/europepmc/webservices/rest/search?query=DOI:{doi}&format=json'
        req = urllib.request.Request(url, headers={'User-Agent': 'Mozilla/5.0'})
        with urllib.request.urlopen(req, timeout=8) as r:
            data = json.loads(r.read())
        result_list = data.get('resultList') or {}  # ← None-safe
        results = result_list.get('result') or []  # ← None-safe
        if results and results[0].get('pmcid'):
            return f'https://europepmc.org/articles/{results[0]["pmcid"]}?pdf=render'
    except Exception:
        pass
    return None

# 跑
for tag, cand in safe_unpaywall_api(doi):
    data, src = safe_try_download(tab, cand, timeout=25)
    if data and data[:4] == b'%PDF' and verify_pdf_content(data, doi, dst_path):
        return save_pdf(data, dst_path, rec)

epmc = safe_epmc_pdf(doi)
if epmc:
    data, src = safe_try_download(tab, epmc, timeout=25)
    if data and data[:4] == b'%PDF' and verify_pdf_content(data, doi, dst_path):
        return save_pdf(data, dst_path, rec)
```

**关键 bug** (v1.5.0 真实踩坑):
- v13 `unpaywall_api` 当 `best_oa_location == null` 时, `best.get('url_for_pdf')` 抛 AttributeError
- v13 `epmc_pdf` 当 `resultList == null` 时, `data.get('resultList', {}).get('result', [])` 抛 AttributeError

**修法**: `or {}` / `or []` 把 None 变空 dict/list.

```python
        new_pdfs = dw.check_new()
        if new_pdfs:

                rec['source'] = f'scihub_auto_{mirror}'

        # iframe PDF URL (v13 JS_GET_SCIHUB_PDF)
        pdf_url = tab.eval(JS_GET_SCIHUB_PDF)
        if pdf_url:
            clean_url = pdf_url.split('#')[0]
            data, src = safe_try_download(tab, clean_url, timeout=25)
            if data and isinstance(data, bytes) and data[:4] == b'%PDF':
                if verify_pdf_content(data, doi, dst_path):
                    rec['source'] = f'scihub_{mirror}_{src}'
                    return save_pdf(data, dst_path, rec)
    except Exception as e:
        rec.setdefault('scihub_errors', []).append(f'{mirror}: {e!r}')
        continue
```

**JS_GET_SCIHUB_PDF** (v13 已有):
```javascript
let pdfUrl = null;
for (let f of document.querySelectorAll('iframe')) {
    if (f.src && /\.pdf($|\?|#)/.test(f.src)) {
        pdfUrl = f.src; break;
    }
}
if (!pdfUrl) {
    let emb = document.querySelector("embed[type='application/pdf'], object[type='application/pdf']");
    if (emb) pdfUrl = emb.src || emb.data;
}
return pdfUrl;
```

## Step 8: Google filetype:pdf fallback (用户硬要求)

```python
def google_filetype_pdf(tab, query, doi='', max_attempts=3):
    """Google 'filetype:pdf <query>' 搜索, 多 query 变体"""
    queries = []
    if query:
        queries.append(query)
        if 'et al.' in query:
            queries.append(query.split('et al.')[0].strip())
        if '.' in query:
            queries.append(query.split('.')[0])
        queries.append(re.sub(r'\([^)]*\)', '', query).strip())
    
    seen_q = set()
    for q in queries[:max_attempts]:
        if not q or q in seen_q:
            continue
        seen_q.add(q)
        search_url = f'https://www.google.com/search?q={urllib.parse.quote(q)}+filetype:pdf'
        try:
            tab.send('Page.navigate', {'url': search_url})
            time.sleep(5)
        except Exception:
            continue
        
        js = '''
        (() => {
            const links = Array.from(document.querySelectorAll('a[href]'))
                .map(a => a.href)
                .filter(h => h.toLowerCase().includes('.pdf') && !h.includes('google.com'));
            return JSON.stringify(links.slice(0, 5));
        })()
        '''
        try:
            links_str = tab.eval(js)
        except Exception:
            continue
        if not links_str:
            continue
        try:
            links = json.loads(links_str)
        except Exception:
            continue
        
        for link in links:
            data, src = safe_try_download(tab, link, timeout=15)
            if data and isinstance(data, bytes) and data[:4] == b'%PDF':
                if verify_pdf_content(data, doi, q):
                    return data, f'google_q={q[:30]}_{src}'
    return None, 'no_valid_pdf_in_google'
```
