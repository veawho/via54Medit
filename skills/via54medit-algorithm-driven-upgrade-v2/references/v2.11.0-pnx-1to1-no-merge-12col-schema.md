def find_target(name, visited=None):
    if visited is None: visited = set()
    if name in visited: return name
    visited.add(name)
    if name in merge_map and merge_map[name] != name:
        return find_target(merge_map[name], visited)
    return name

for old, new in merge_map.items():
    final_targets[old] = find_target(new)

# idempotent move + 同名文件加后缀
for old, new in to_merge.items():
    src = f'{base}/{old}'
    dst = f'{base}/{new}'
    if os.path.isdir(src):
        if not os.path.isdir(dst):
            os.rename(src, dst)
        else:
            for f in os.listdir(src):
                src_file = f'{src}/{f}'
                dst_file = f'{dst}/{f}'
                if not os.path.exists(dst_file):
                    shutil.move(src_file, dst_file)
                else:
                    base, ext = os.path.splitext(f)
                    i = 1
                    while os.path.exists(f'{dst}/{base}_{i}{ext}'):
                        i += 1
                    shutil.move(src_file, f'{dst}/{base}_{i}{ext}')
            try:
                os.rmdir(src)
            except: pass

def highlight_pdf(pdf_path, keywords, max_pages=20):
    doc = fitz.open(pdf_path)
    matches = []
    for page_num in range(min(max_pages, len(doc))):
        page = doc[page_num]
        for kw in keywords:
            text_instances = page.search_for(kw)
            for inst in text_instances:
                highlight = page.add_highlight_annot(inst)
                highlight.set_colors(stroke=(1, 1, 0))  # 黄色
                highlight.update()
                matches.append((page_num + 1, kw, inst))
    doc.saveIncr()
    doc.close()
    return pdf_path, len(matches)

def render_highlighted_pages(pdf_path, out_dir, max_pages=5, dpi=120):
