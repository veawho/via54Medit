## v6 算法 (主入口)

**`scripts/analyze_ppt_citations_v6_vision_align.py`** — D 列 ≥50% 命中度量

```python
#!/usr/bin/env python3
import re, csv, json

MISALIGNED_4 = {('12', '5'), ('14', '2'), ('22', '13'), ('30', '10')}

def evaluate(truth, all_v):
    total_pct = total_yue = total_wan = 0
    hit_pct = hit_yue = hit_wan = 0
    hit_marks = 0
    miss = []
    for r in truth:
        a, n = r['﻿PPT页'], r['第几条']
        if (a, n) in MISALIGNED_4: continue
        truth_d = r['引用语义（上下文）']
        truth_pcts = set(re.findall(r'\d+\.?\d*%', truth_d))
        truth_yue = set(re.findall(r'\d+\.?\d*月', truth_d))
        truth_wan = set(re.findall(r'\d+\.?\d*万', truth_d))
        my = all_v.get(a, {}).get(n, {})
        my_d = my.get('data_points', []) if isinstance(my, dict) else []
        my_pcts = set(p for p in my_d if '%' in p)
        my_yue = set(p for p in my_d if '月' in p)
        my_wan = set(p for p in my_d if '万' in p)
        total_pct += len(truth_pcts); hit_pct += len(truth_pcts & my_pcts)
        total_yue += len(truth_yue); hit_yue += len(truth_yue & my_yue)
        total_wan += len(truth_wan); hit_wan += len(truth_wan & my_wan)
        total_truth = len(truth_pcts) + len(truth_yue) + len(truth_wan)
        total_hit = len(truth_pcts & my_pcts) + len(truth_yue & my_yue) + len(truth_wan & my_wan)
        if total_hit / max(1, total_truth) >= 0.5:
            hit_marks += 1
        else:
            miss.append((a, n, total_truth, total_hit))
    return {
        'pct': hit_pct/max(1,total_pct)*100,
        'yue': hit_yue/max(1,total_yue)*100,
        'wan': hit_wan/max(1,total_wan)*100,
        'hit_marks': hit_marks,
        'miss': miss
    }

# 主流程
truth = []
with open('/Users/david/Desktop/雷管方案_文献整理/_citation_table/citation_table.csv') as f:
    truth = [r for r in csv.DictReader(f) if (r['﻿PPT页'], r['第几条']) not in MISALIGNED_4]

with open('/tmp/vision_d_merged.json') as f:
    all_v = json.load(f)

result = evaluate(truth, all_v)
print(f'≥50% 命中: {result["hit_marks"]}/160 = {result["hit_marks"]/160*100:.1f}%')
print(f'% 覆盖: {result["pct"]:.1f}%')
print(f'月 覆盖: {result["yue"]:.1f}%')
print(f'万 覆盖: {result["wan"]:.1f}%')
print(f'漏: {len(result["miss"])}')
```

### 4. 合并 + 拆纯数字

合并脚本必须兼容 6 种格式 (dict of dict / list of dicts / 散文 / P_N 嵌套 / mark_N 顶层 / mark_N 散文).

```python
import re

# 拆纯数字
for s, marks in all_v.items():
    for n, info in marks.items():
        if not isinstance(info, dict): continue
        data = info.get('data_points', [])
        if not data: continue
        new_data = list(data)
        for p in data:
            m = re.match(r'^(.*?)(\d+\.?\d*\s*%|\d+\.?\d*\s*月|\d+\.?\d*\s*万)(.*)$', p)
            if m:
                num_match = re.search(r'\d+\.?\d*', m.group(2))
                if num_match:
                    num_str = num_match.group(0)
                    if '%' in m.group(2): num_str += '%'
                    elif '月' in m.group(2): num_str += '月'
                    elif '万' in m.group(2): num_str += '万'
                    if num_str not in new_data:
                        new_data.append(num_str)
        info['data_points'] = new_data
```

## 子 agent JSON 6 种格式兼容

| 格式 | 触发 | 处理 |
|------|------|------|
| 1 | `{slide_N: {mark_N: {...}}}` dict of dict | 直接展开 |
| 2 | `[{slide: N, img: ..., marks_observed: [{mark, position: {x_c, y_c}, data_points, ...}]}]` list of dicts | 遍历 marks_observed |
| 3 | `{slide_N: "散文", P26: "...", ...}` str values | 退化成 metadata, 不提取 |
| 4 | `{marks: [{mark, all_numbers_on_ppt: [{value, unit, context}]}]}` (P5 格式) | 提 all_numbers_on_ppt value+unit |
| 5 | `{slide_N: {mark_1_3_5_specific_data_points_summary: {mark_N: {attached_data_points: [{where, units: [...]}]}}}}` (P4 格式) | 提 attached_data_points.units |
| 6 | 顶层 `mark_N: {position, shape_type, data_points, context, related_visual}` (P26/27) | 直接展开 |

**实战坑**: 不兼容 6 种格式 → 漏 60% 数据点.
